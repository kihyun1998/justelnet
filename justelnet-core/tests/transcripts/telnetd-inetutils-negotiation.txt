# authentication, against the default policy. The order is the source's: the
# phase 1 DOs go out, then the server reads the Core's active start and
# answers it, and send_do/send_will skip any option already on, so phase 2
# holds only what is still unsettled. (The same answer order was seen on a
# netkit telnetd, telnetd-netkit-rhel9.txt.)
# The real server, recorded, is telnetd-inetutils-2.8-recorded.txt; it also

# Phase 1: DO TTYPE, TSPEED, XDISPLOC, NEW-ENVIRON, OLD-ENVIRON. TTYPE and
# NEW-ENVIRON agree with what the Core already offered; the rest are refused,

client: ff fc 20 ff fc 23 ff fc 24

event: option local NEW-ENVIRON on

# Answers to the active start: DO NAWS, WILL ECHO, DO SGA, WILL SGA.

client: ff fa 1f 00 50 00 18 ff f0

event: option remote ECHO on
event: option local SGA on
event: option remote SGA on
